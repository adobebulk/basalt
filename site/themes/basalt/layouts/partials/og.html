{{/* Open Graph / Twitter card tags. IMPORTANT: og:image points at the JPEG variant (1200.jpg), never AVIF — most link unfurlers don't render AVIF. assetsBaseURL is root-relative (/assets), so we use absURL to produce the absolute https://photos.ctsmith.org/assets/... URL required by OG/Twitter. */ -}} {{- $base := .Site.Params.assetsBaseURL -}} {{- $img := "" -}} {{- $title := .Title -}} {{- $desc := "" -}} {{- if .Params.photoid -}} {{- $photo := partial "getphoto.html" . -}} {{- with $photo -}} {{- $img = partial "asset-url.html" (dict "photo" . "file" "1200.jpg") | absURL -}} {{- $cap := string .caption -}} {{- $title = cond (ne $cap "") $cap $.Parent.Title -}} {{- $desc = $.Parent.Params.description -}} {{- end -}} {{- else if .Params.photos -}} {{- $cover := .Params.cover -}} {{- $cphoto := dict -}} {{- range .Params.photos -}}{{- if eq .id $cover -}}{{- $cphoto = . -}}{{- end -}}{{- end -}} {{- if and (not $cphoto.key) (gt (len .Params.photos) 0) -}}{{- $cphoto = index .Params.photos 0 -}}{{- end -}} {{- if $cphoto.key -}}{{- $img = partial "asset-url.html" (dict "photo" $cphoto "file" "1200.jpg") | absURL -}}{{- end -}} {{- $desc = .Params.description -}} {{- else -}} {{- $desc = .Site.Params.description -}} {{- end -}} {{- with $desc }} {{ end }} {{- with $img }} {{ end }}
